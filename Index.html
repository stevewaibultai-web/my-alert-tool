import time import requests # 🤖 TELEGRAM CONFIGURATION (Replace with your actual data later) TOKEN = "YOUR_BOT_TOKEN_HERE" CHAT_ID = "YOUR_CHAT_ID_HERE" # 📊 TRACKED SYMBOLS SETUP TRACKED_SYMBOLS = [ {"symbol": "BINANCE:BTCUSDT", "indicator": "MACD", "timeframe": "15m", "status": "ON"}, {"symbol": "BINANCE:ETHUSDT", "indicator": "EMA", "timeframe": "30m", "status": "ON"}, {"symbol": "NSE:RELIANCE", "indicator": "RSI", "timeframe": "1h", "status": "ON"}, {"symbol": "NSE:TCS", "indicator": "MACD", "timeframe": "4h", "status": "ON"} ] # Duplicate Notification Filter Memory last_alerted_state = {} def send_telegram_message(text): """Function to send instant alert to Telegram Bot""" url = f"https://telegram.org{TOKEN}/sendMessage" payload = {"chat_id": CHAT_ID, "text": text} try: requests.post(url, json=payload, timeout=10) except Exception as e: print("Telegram error:", e) def fetch_tradingview_data(tv_symbol): """Function to fetch technical data directly from TradingView Scanner""" url = "https://tradingview.com" payload = { "symbols": {"tickers": [tv_symbol], "query": {"types": []}}, "columns": ["close", "RSI", "MACD.macd", "MACD.signal", "EMA20", "EMA50"] } try: response = requests.post(url, json=payload, timeout=15) data = response.json() if "data" in data and len(data["data"]) > 0: return data["data"][0]["d"] except Exception as e: print(f"Error fetching data for {tv_symbol}: {e}") return None def process_alerts(): """Main indicator crossover logic controller""" for item in TRACKED_SYMBOLS: if item["status"] == "OFF": continue symbol = item["symbol"] indicator = item["indicator"] tf = item["timeframe"] alert_key = f"{symbol}_{indicator}_{tf}" data_values = fetch_tradingview_data(symbol) if not data_values: continue current_price = data_values[0] rsi_val = data_values[1] macd_main = data_values[2] macd_signal = data_values[3] ema20 = data_values[4] ema50 = data_values[5] message = "" # 1. RSI Alert Logic if indicator == "RSI" and rsi_val: if rsi_val >= 70 and last_alerted_state.get(alert_key) != "overbought": message = f"🚨 RSI Overbought! \n🪙 Symbol: {symbol}\n⏱️ TF: {tf}\n💰 Price: {current_price}\n📊 RSI: {rsi_val:.2f}" last_alerted_state[alert_key] = "overbought" elif rsi_val <= 30 and last_alerted_state.get(alert_key) != "oversold": message = f"🚨 RSI Oversold! \n🪙 Symbol: {symbol}\n⏱️ TF: {tf}\n💰 Price: {current_price}\n📊 RSI: {rsi_val:.2f}" last_alerted_state[alert_key] = "oversold" elif 30 < rsi_val < 70: last_alerted_state[alert_key] = "normal" # 2. MACD Crossover Alert Logic elif indicator == "MACD" and macd_main and macd_signal: current_cross = "bullish" if macd_main > macd_signal else "bearish" if alert_key in last_alerted_state and last_alerted_state[alert_key] != current_cross: message = f"🔔 MACD Crossover! \n🪙 Symbol: {symbol}\n⏱️ TF: {tf}\n📈 Trend: {current_cross.upper()}\n💰 Price: {current_price}" last_alerted_state[alert_key] = current_cross # 3. EMA (20/50) Crossover Alert Logic elif indicator == "EMA" and ema20 and ema50: current_cross = "bullish" if ema20 > ema50 else "bearish" if alert_key in last_alerted_state and last_alerted_state[alert_key] != current_cross: message = f"📈 EMA Crossover (20/50)! \n🪙 Symbol: {symbol}\n⏱️ TF: {tf}\n🚀 Signal: {current_cross.upper()}\n💰 Price: {current_price}" last_alerted_state[alert_key] = current_cross if message: send_telegram_message(message) # Loop to check every 10 seconds background if __name__ == "__main__": print("🚀 Trading Bot Background Server Started Successfully...") send_telegram_message("🚀 Trading Bot Server Started Successfully! Notifications are live now.") while True: process_alerts() time.sleep(10)
